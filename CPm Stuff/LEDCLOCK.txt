; ==============================================================
; TIME719Q.ASM — 8080 mnemonics, CP/M + RomWBW HBIOS + SC719
; --------------------------------------------------------------
; Features:
;   * Reads the real-time clock (SC727 module) via RomWBW HBIOS
;   * Displays date/time at top left of the console, one line only
;   * Updates every second without scrolling
;   * Mirrors the seconds value (0–59) to the SC719 LED outputs
;   * Press Q or q to quit: LEDs cleared, cursor restored
;
; Hardware:
;   - RTC: SC727 RTC module at I/O base C0h (handled by HBIOS)
;   - LED: SC719 Digital I/O module, base address selectable via JP1
;
; Notes:
;   - All HBIOS device access is via CALL RST 1 (vector 0008h).
;   - The RTC is not accessed directly by port; HBIOS abstracts it.
;   - The SC719 LEDs are mapped to a base I/O port. Default: 00h.
; ===============================================================

BDOS        EQU 0005H        ; CP/M BDOS entry
CONOUT      EQU 2            ; BDOS function 2: console output

; ====== CONFIGURE YOUR SC719 I/O BASE ADDRESS HERE ======
; JP1 on SC719 sets the base I/O port. Default with no jumpers = 00h.
; Examples: DIO_BASE EQU 00H (default), 01H, 20H, A0H, etc.
DIO_BASE    EQU 00H

; LED polarity: 00H if active-high (LED on = logic 1),
;               0FFH if active-low (LED on = logic 0).
INVERT_XOR  EQU 00H

            ORG 0100H        ; CP/M COM files start at 0100h

; ===============================================================
; Program Entry
; ===============================================================
START:
            ; Clear screen, home cursor, hide cursor
            LXI  D,ESC2J     ; ESC[2J$  (clear screen)
            MVI  C,9
            CALL BDOS
            LXI  D,HOME      ; ESC[H$   (cursor home)
            MVI  C,9
            CALL BDOS
            LXI  D,HIDECUR   ; ESC[?25l$ (hide cursor)
            MVI  C,9
            CALL BDOS

            ; --- LED self-test: pattern to verify wiring & address ---
            MVI  A,055H      ; 01010101
            CALL LEDOUT
            CALL SHORTDLY
            MVI  A,0AAH      ; 10101010
            CALL LEDOUT
            CALL SHORTDLY
            MVI  A,0FFH      ; all on
            CALL LEDOUT
            CALL SHORTDLY
            XRA  A           ; 00h = all off
            CALL LEDOUT
            CALL SHORTDLY

; ===============================================================
; Main loop: update display and LEDs once per second
; ===============================================================
MAIN:
            ; Reset cursor to top-left each cycle
            LXI  D,HOME
            MVI  C,9
            CALL BDOS

            CALL READRTC             ; fetch current RTC values
            LDA  TIMBUF+5            ; seconds (BCD)
            STA  SEC0                ; remember for tick change

            CALL PRINTDT             ; print formatted date/time
            CALL WRITELED            ; send seconds to LEDs

; --- Wait for seconds to change before repeating ---
WAITNEXT:
            CALL KEYCHECK            ; check keyboard
            ORA  A
            JZ   WN1
            ANI  5FH                 ; normalize to uppercase
            CPI  'Q'                 ; quit key?
            JZ   QUIT
WN1:
            CALL SHORTDLY
            CALL READRTC
            LDA  TIMBUF+5            ; new seconds (BCD)
            MOV  D,A                 ; save in D
            LDA  SEC0                ; old seconds
            CMP  D
            JZ   WAITNEXT            ; still same second → keep waiting
            MOV  A,D
            STA  SEC0                ; update
            JMP  MAIN

; ===============================================================
; Quit sequence
; ===============================================================
QUIT:
            XRA  A
            CALL LEDOUT              ; clear LEDs
            LXI  D,SHOWCUR           ; restore cursor
            MVI  C,9
            CALL BDOS
            MVI  E,13
            CALL PCHAR               ; CR
            MVI  E,10
            CALL PCHAR               ; LF
            RET

; ===============================================================
; HBIOS RTC Access
; ===============================================================
; HBIOS function: RTCGETTIM (B=20h, HL=buffer, RST 1)
; Returns 6 BCD bytes in buffer: YY,MM,DD,HH,MM,SS
; (RomWBW handles actual RTC I/O at port C0h internally)
; ===============================================================
READRTC:    LXI  H,TIMBUF
            MVI  B,020H
            RST  1
            RET

; ===============================================================
; Display date/time: YYYY-MM-DD HH:MM:SS
; ===============================================================
PRINTDT:    MVI  E,'2'
            CALL PCHAR
            MVI  E,'0'
            CALL PCHAR
            LDA  TIMBUF+0            ; year (YY)
            CALL P2DIG
            MVI  E,'-'
            CALL PCHAR
            LDA  TIMBUF+1            ; month
            CALL P2DIG
            MVI  E,'-'
            CALL PCHAR
            LDA  TIMBUF+2            ; day
            CALL P2DIG
            MVI  E,' '
            CALL PCHAR
            LDA  TIMBUF+3            ; hour
            CALL P2DIG
            MVI  E,':'
            CALL PCHAR
            LDA  TIMBUF+4            ; minute
            CALL P2DIG
            MVI  E,':'
            CALL PCHAR
            LDA  TIMBUF+5            ; second
            CALL P2DIG
            RET

; ===============================================================
; Write seconds to SC719 LEDs
; ===============================================================
; TIMBUF+5 holds seconds in BCD: e.g. 0x25 = 25 sec.
; Convert to binary 0..59 and output to SC719.
; ===============================================================
WRITELED:
            LDA  TIMBUF+5            ; BCD seconds
            MOV  D,A
            ANI  0FH                 ; ones digit
            MOV  E,A                 ; E = ones
            MOV  A,D
            ANI  0F0H                ; tens nibble
            RRC
            RRC
            RRC
            RRC                      ; A = tens
            MOV  B,A                 ; B = tens
            ; Compute seconds = tens*10 + ones
            MOV  A,B
            ADD  A                   ; A = 2*t
            MOV  C,A
            MOV  A,B
            ADD  A                   ; 2*t
            ADD  A                   ; 4*t
            ADD  A                   ; 8*t
            ADD  C                   ; 10*t
            ADD  E                   ; + ones
            ; fallthrough to LEDOUT with A

; ===============================================================
; LEDOUT: write A to SC719 outputs
; ===============================================================
; SC719 base I/O port = DIO_BASE (JP1 selectable).
; OUT writes directly to 8-bit latch driving LEDs.
; ===============================================================
LEDOUT:
            XRI  INVERT_XOR          ; apply optional inversion
            OUT  DIO_BASE            ; send to SC719
            RET

; ===============================================================
; KEYCHECK: non-blocking key read (BDOS 6, E=0FFh)
; Returns: A=0 if no key; else ASCII char
; ===============================================================
KEYCHECK:
            PUSH B
            PUSH D
            MVI  C,6
            MVI  E,0FFH
            CALL BDOS
            MOV  L,A
            POP  D
            POP  B
            MOV  A,L
            RET

; ===============================================================
; Helper routines: BCD→ASCII, Console output, Delay
; ===============================================================
P2DIG:      CALL BCD2ASC             ; A=BCD -> HL=ASCII digits
            MOV  A,H
            STA  DIGTMP
            MOV  A,L
            STA  DIGTMP+1
            LDA  DIGTMP
            MOV  E,A
            CALL PCHAR
            LDA  DIGTMP+1
            MOV  E,A
            CALL PCHAR
            RET

BCD2ASC:    MOV  D,A
            ANI  0FH
            ADI  '0'
            MOV  L,A
            MOV  A,D
            ANI  0F0H
            RRC
            RRC
            RRC
            RRC
            ADI  '0'
            MOV  H,A
            RET

PCHAR:      MVI  C,CONOUT
            CALL BDOS
            RET

SHORTDLY:   LXI  H,2000H             ; short delay
SD1:        DCX  H
            MOV  A,H
            ORA  L
            JNZ  SD1
            RET

; ===============================================================
; Data storage
; ===============================================================
TIMBUF:     DS   6                   ; RTC values YY,MM,DD,HH,MM,SS
DIGTMP:     DS   2                   ; temp buffer for ASCII
SEC0:       DB   00H                 ; previous seconds

; ===============================================================
; ANSI escape sequences ($-terminated for BDOS 9)
; ===============================================================
ESC2J:      DB  1BH,'[','2','J','$'     ; clear screen
HOME:       DB  1BH,'[','H','$'         ; cursor home
HIDECUR:    DB  1BH,'?','2','5','l','$' ; hide cursor
SHOWCUR:    DB  1BH,'?','2','5','h','$' ; show cursor

            END START
