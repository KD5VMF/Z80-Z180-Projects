; TIME719Q.ASM — 8080 mnemonics, CP/M + RomWBW HBIOS + SC719
; Top-left clock (one line), LEDs show seconds on SC719 @ DIO_BASE.
; Q or q quits (restores cursor, LEDs off).

BDOS        EQU 0005H
CONOUT      EQU 2

; ====== CONFIGURE YOUR SC719 I/O BASE ADDRESS HERE ======
; JP1 on SC719 sets the base I/O port; typical default is 00h.
DIO_BASE    EQU 00H

; LED polarity: 00H for active-high, 0FFH for active-low
INVERT_XOR  EQU 00H

            ORG 0100H

START:
            ; Clear + home + hide cursor
            LXI  D,ESC2J
            MVI  C,9
            CALL BDOS
            LXI  D,HOME
            MVI  C,9
            CALL BDOS
            LXI  D,HIDECUR
            MVI  C,9
            CALL BDOS

            ; Self-test LEDs: 55 -> AA -> FF -> 00
            MVI  A,055H
            CALL LEDOUT
            CALL SHORTDLY
            MVI  A,0AAH
            CALL LEDOUT
            CALL SHORTDLY
            MVI  A,0FFH
            CALL LEDOUT
            CALL SHORTDLY
            XRA  A
            CALL LEDOUT
            CALL SHORTDLY

MAIN:
            ; Home cursor for in-place update
            LXI  D,HOME
            MVI  C,9
            CALL BDOS

            CALL READRTC            ; TIMBUF = BCD YY MM DD HH MM SS
            LDA  TIMBUF+5
            STA  SEC0               ; remember current seconds

            CALL PRINTDT            ; show YYYY-MM-DD HH:MM:SS (no CR/LF)
            CALL WRITELED           ; mirror seconds to LEDs @ DIO_BASE

WAITNEXT:
            ; non-blocking quit?
            CALL KEYCHECK
            ORA  A
            JZ   WN1
            ANI  5FH                ; uppercase
            CPI  'Q'
            JZ   QUIT
WN1:
            CALL SHORTDLY
            CALL READRTC
            LDA  TIMBUF+5           ; A = current seconds (BCD)
            MOV  D,A                ; save current in D
            LDA  SEC0               ; A = previous seconds (BCD)
            CMP  D                  ; changed yet?
            JZ   WAITNEXT
            MOV  A,D
            STA  SEC0
            JMP  MAIN

QUIT:
            XRA  A                  ; LEDs off (handled via LEDOUT inversion)
            CALL LEDOUT
            LXI  D,SHOWCUR          ; show cursor
            MVI  C,9
            CALL BDOS
            MVI  E,13               ; CR
            CALL PCHAR
            MVI  E,10               ; LF
            CALL PCHAR
            RET

; --- HBIOS RTCGETTIM (B=20h, HL=buffer), entry 0008h (RST 1) ---
; Returns TIMBUF = YY,MM,DD,HH,MM,SS  (all BCD)
READRTC:    LXI  H,TIMBUF
            MVI  B,020H
            RST  1
            RET

; --- Print YYYY-MM-DD HH:MM:SS (no CR/LF) ---
PRINTDT:    MVI  E,'2'
            CALL PCHAR
            MVI  E,'0'
            CALL PCHAR
            LDA  TIMBUF+0           ; YY
            CALL P2DIG
            MVI  E,'-'
            CALL PCHAR
            LDA  TIMBUF+1           ; MM
            CALL P2DIG
            MVI  E,'-'
            CALL PCHAR
            LDA  TIMBUF+2           ; DD
            CALL P2DIG
            MVI  E,' '
            CALL PCHAR
            LDA  TIMBUF+3           ; HH
            CALL P2DIG
            MVI  E,':'
            CALL PCHAR
            LDA  TIMBUF+4           ; MM
            CALL P2DIG
            MVI  E,':'
            CALL PCHAR
            LDA  TIMBUF+5           ; SS
            CALL P2DIG
            RET

; --- Seconds to LEDs: BCD SS -> binary 0..59, then OUT DIO_BASE ---
WRITELED:
            LDA  TIMBUF+5           ; BCD seconds
            MOV  D,A
            ANI  00FH               ; ones -> A
            MOV  E,A                ; E = ones
            MOV  A,D
            ANI  0F0H               ; tens * 16
            RRC
            RRC
            RRC
            RRC                     ; A = tens (0..5)
            MOV  B,A                ; B = tens
            ; A = tens*10 = (t<<3) + (t<<1)
            MOV  A,B
            ADD  A                  ; 2*t
            MOV  C,A                ; C = 2*t
            MOV  A,B
            ADD  A                  ; 2*t
            ADD  A                  ; 4*t
            ADD  A                  ; 8*t
            ADD  C                  ; 10*t
            ADD  E                  ; + ones
            ; fallthrough to LEDOUT with A

; --- LEDOUT: write A to SC719 (applies optional inversion) ---
LEDOUT:
            XRI  INVERT_XOR         ; set to 0FFH if active-low LEDs
            OUT  DIO_BASE           ; <-- SC719 base I/O address (JP1)
            RET

; --- Non-blocking console read: BDOS 6, E=FFh (A=char or 0) ---
KEYCHECK:
            PUSH B
            PUSH D
            MVI  C,6
            MVI  E,0FFH
            CALL BDOS
            MOV  L,A
            POP  D
            POP  B
            MOV  A,L
            RET

; --- A=BCD -> print two ASCII digits (protect from BDOS clobber) ---
P2DIG:      CALL BCD2ASC            ; H=tens ASCII, L=ones ASCII
            MOV  A,H
            STA  DIGTMP
            MOV  A,L
            STA  DIGTMP+1
            LDA  DIGTMP
            MOV  E,A
            CALL PCHAR
            LDA  DIGTMP+1
            MOV  E,A
            CALL PCHAR
            RET

; --- BCD -> ASCII pair in H,L ---
BCD2ASC:    MOV  D,A
            ANI  00FH               ; ones
            ADI  '0'
            MOV  L,A
            MOV  A,D
            ANI  0F0H               ; tens
            RRC
            RRC
            RRC
            RRC
            ADI  '0'
            MOV  H,A
            RET

; --- BDOS char out (E) ---
PCHAR:      MVI  C,CONOUT
            CALL BDOS
            RET

; --- short delay (self-test & tick wait idle) ---
SHORTDLY:   LXI  H,2000H
SD1:        DCX  H
            MOV  A,H
            ORA  L
            JNZ  SD1
            RET

; --- Data ---
TIMBUF:     DS   6                  ; YY MM DD HH MM SS (BCD)
DIGTMP:     DS   2
SEC0:       DB   00H

; --- ANSI escape strings (BDOS 9, $-terminated) ---
ESC2J:      DB  1BH,'[','2','J','$'     ; clear screen
HOME:       DB  1BH,'[','H','$'         ; cursor to 1;1
HIDECUR:    DB  1BH,'?','2','5','l','$' ; hide cursor
SHOWCUR:    DB  1BH,'?','2','5','h','$' ; show cursor

            END START
