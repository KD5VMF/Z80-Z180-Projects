; ============================================================================
; PRIMEGEN.ASM — 8080 / CP/M
; Prints primes forever (numbers only). Stop with ^C.
; Trial division by odd D; stop testing when D*D > N.
; Each prime printed with one BDOS 9 call (string at DE, '$' terminated).
;
; Now prints with thousands separators (commas).
;
; ============================================================================
BDOS    EQU     0005H

            ORG     0100H

; ------------------------
; ENTRY
; ------------------------
START:
            LXI     D, TITLE
            MVI     C, 9
            CALL    BDOS

            ; N = 2, print it
            MVI     A, 2
            STA     N0
            XRA     A
            STA     N1
            STA     N2
            STA     N3

            LXI     H, N0
            SHLD    PTR32
            CALL    U32_PRINT_PTR_BDOS9
            LXI     D, NL
            MVI     C, 9
            CALL    BDOS

            ; N = 3 (start odd sequence)
            MVI     A, 3
            STA     N0
            XRA     A
            STA     N1
            STA     N2
            STA     N3

; ------------------------
; MAIN LOOP  (no key checks; use ^C to stop)
; ------------------------
MAIN_LOOP:
            ; Trial divisors start at 3
            MVI     D, 0
            MVI     E, 3          ; DE = 3

DIV_LOOP:
            ; If (D*D) > N ⇒ N is prime
            PUSH    D
            PUSH    B
            PUSH    H
            CALL    SQR16_T       ; T = DE * DE
            CALL    T_CMP_N       ; CY=1 if T<N, Z=1 if T==N, else T>N
            POP     H
            POP     B
            POP     D
            JC      TRY_DIV       ; T <= N ? keep testing divisors
            JZ      TRY_DIV

            ; prime: print N
            LXI     H, N0
            SHLD    PTR32
            CALL    U32_PRINT_PTR_BDOS9
            LXI     D, NL
            MVI     C, 9
            CALL    BDOS
            JMP     NEXT_N

TRY_DIV:    ; If N % D == 0 ⇒ composite
            PUSH    D
            CALL    U32_MOD_DE    ; remainder in HL
            MOV     A, H
            ORA     L
            POP     D
            JZ      NEXT_N        ; composite ⇒ advance N

            ; D += 2 (odd divisors only)
            MVI     A, 2
            ADD     E
            MOV     E, A
            JNC     DIV_LOOP
            INR     D
            JMP     DIV_LOOP

NEXT_N:     ; N += 2 (skip evens)
            MVI     A, 2
            CALL    ADD_N_U8
            JMP     MAIN_LOOP

; ------------------------
; CLEAN EXIT (unused; ^C terminates)
; ------------------------
CLEAN_EXIT:
            RET

; ============================================================================
; UTILITIES
; ============================================================================

; Add 8-bit A to 32-bit N (N0..N3)
ADD_N_U8:
            MOV     E, A
            LDA     N0
            ADD     E
            STA     N0
            LDA     N1
            ACI     0
            STA     N1
            LDA     N2
            ACI     0
            STA     N2
            LDA     N3
            ACI     0
            STA     N3
            RET

; ------------------------
; Decimal print of u32 at PTR32 (single BDOS 9 call)
;   - Builds digits (with commas) into OUTBUF[..OUTEND-1], puts '$' at OUTEND
;   - Calls BDOS 9 once with DE = start of digits
; ------------------------
U32_PRINT_PTR_BDOS9:
            ; Copy source (4 bytes) into NW0..NW3
            LHLD    PTR32
            MOV     A, M
            STA     NW0
            INX     H
            MOV     A, M
            STA     NW1
            INX     H
            MOV     A, M
            STA     NW2
            INX     H
            MOV     A, M
            STA     NW3

            ; If zero -> print "0$"
            LDA     NW3
            MOV     B, A
            LDA     NW2
            ORA     B
            MOV     B, A
            LDA     NW1
            ORA     B
            MOV     B, A
            LDA     NW0
            ORA     B
            MOV     A, B
            JNZ     U32P_CONT9

            MVI     A, '0'
            STA     OUTBUF+0
            MVI     A, '$'
            STA     OUTBUF+1
            LXI     D, OUTBUF
            MVI     C, 9
            CALL    BDOS
            RET

U32P_CONT9:
            ; Prepare buffer tail and write terminator
            LXI     H, OUTEND
            SHLD    OUTPTR
            MVI     A, '$'
            STA     OUTEND
            XRA     A
            STA     DIGCNT         ; NEW: digit counter since last comma (0..3)

U32P_LOOP9:
            CALL    DIV32_10       ; NW /= 10, remainder -> B

            ; store digit at [--OUTPTR]
            LHLD    OUTPTR
            DCX     H
            MOV     A, B
            ADI     '0'
            MOV     M, A
            SHLD    OUTPTR

            ; increment digit counter
            LDA     DIGCNT
            INR     A
            STA     DIGCNT

            ; if 3 digits collected and more digits remain, insert comma
            CPI     3
            JNZ     U32P_NZCHK

            ; check if NW != 0 (more digits to the left)
            LDA     NW3
            MOV     B, A
            LDA     NW2
            ORA     B
            MOV     B, A
            LDA     NW1
            ORA     B
            MOV     B, A
            LDA     NW0
            ORA     B
            MOV     A, B
            JZ      U32P_NZCHK     ; no more digits → no comma

            ; insert comma
            LHLD    OUTPTR
            DCX     H
            MVI     M, ','
            SHLD    OUTPTR
            XRA     A
            STA     DIGCNT         ; reset counter to 0

U32P_NZCHK:
            ; if NW != 0 keep collecting
            LDA     NW3
            MOV     B, A
            LDA     NW2
            ORA     B
            MOV     B, A
            LDA     NW1
            ORA     B
            MOV     B, A
            LDA     NW0
            ORA     B
            MOV     A, B
            JNZ     U32P_LOOP9

            ; Print from OUTPTR to '$' with one BDOS 9 call
            LHLD    OUTPTR
            XCHG                    ; DE = start
            MVI     C, 9
            CALL    BDOS
            RET

; Divide NW(32-bit) by 10.
; Returns: quotient in NW0..NW3, remainder (0..9) in B.
DIV32_10:
            MVI     B, 0
            LDA     NW3
            CALL    DIV10_STEP
            STA     NW3
            LDA     NW2
            CALL    DIV10_STEP
            STA     NW2
            LDA     NW1
            CALL    DIV10_STEP
            STA     NW1
            LDA     NW0
            CALL    DIV10_STEP
            STA     NW0
            RET

; --- FIXED 16-bit /10 for one byte carry-in ---
; In:  A = low byte, B = remainder_in (0..9)
; Out: A = quotient_byte, B = remainder_out (0..9)
; Uses: C,D,E,B,A (clobbered)
DIV10_STEP:
            MOV     C, A          ; X low
            MOV     A, B
            MOV     D, A          ; X high  (B→D)
            XRA     A
            MOV     E, A          ; Q = 0

; while X >= 160:  X -= 160; Q += 16
D10_160CHK:
            MOV     A, D
            CPI     0
            JNZ     D10_160SUB        ; if high>0, definitely ≥160
            MOV     A, C
            CPI     0A0H
            JC      D10_10CHK         ; low<160 and high==0 → skip
D10_160SUB:
            ; X -= 160 (proper 16-bit subtract with borrow)
            MOV     A, C
            SUI     0A0H
            MOV     C, A
            MOV     A, D
            MVI     B, 0
            SBB     B                 ; D = D - borrow
            MOV     D, A
            ; Q += 16
            MOV     A, E
            ADI     16
            MOV     E, A
            JMP     D10_160CHK

; while X >= 10:   X -= 10; Q += 1
D10_10CHK:
            MOV     A, D
            CPI     0
            JNZ     D10_10SUB         ; high>0 → ≥10
            MOV     A, C
            CPI     10
            JC      D10_DONE
D10_10SUB:
            MOV     A, C
            SUI     10
            MOV     C, A
            MOV     A, D
            MVI     B, 0
            SBB     B                 ; subtract borrow
            MOV     D, A
            INR     E
            JMP     D10_10CHK

D10_DONE:
            MOV     A, E              ; quotient byte
            MOV     B, C              ; remainder (0..9)
            RET

; ------------------------
; u32 % u16 : remainder in HL
; Input: N3..N0 (u32), DE = divisor (u16)
U32_MOD_DE:
            XRA     A
            MOV     H, A
            MOV     L, A
            LDA     N3
            CALL    MOD_STEP
            LDA     N2
            CALL    MOD_STEP
            LDA     N1
            CALL    MOD_STEP
            LDA     N0
            CALL    MOD_STEP
            RET

; MOD_STEP: byte in A, divisor DE, remainder HL
MOD_STEP:
            MOV     H, L          ; r<<8
            MOV     L, A          ; + b
MSL:        MOV     A, H
            CMP     D
            JC      MSD
            JNZ     MSS
            MOV     A, L
            CMP     E
            JC      MSD
MSS:        MOV     A, L          ; r -= DE
            SUB     E
            MOV     L, A
            MOV     A, H
            SBB     D
            MOV     H, A
            JMP     MSL
MSD:        RET

; (DE*DE) -> T0..T3 (32-bit), destroys DE
SQR16_T:
            XRA     A
            STA     T0
            STA     T1
            STA     T2
            STA     T3
            MOV     A, E
            STA     M0
            MOV     A, D
            STA     M1
            XRA     A
            STA     M2
            STA     M3
            MVI     C, 16
SQ1:        MOV     A, E
            ANI     1
            JZ      SQ2
            CALL    T_ADD_M
SQ2:        MOV     A, D
            RAR
            MOV     D, A
            MOV     A, E
            RAR
            MOV     E, A
            STC
            CMC
            LDA     M0
            RAL
            STA     M0
            LDA     M1
            RAL
            STA     M1
            LDA     M2
            RAL
            STA     M2
            LDA     M3
            RAL
            STA     M3
            DCR     C
            JNZ     SQ1
            RET

; T += M (32-bit)
T_ADD_M:
            LDA     T0
            MOV     B, A
            LDA     M0
            ADD     B
            STA     T0
            LDA     T1
            MOV     B, A
            LDA     M1
            ADC     B
            STA     T1
            LDA     T2
            MOV     B, A
            LDA     M2
            ADC     B
            STA     T2
            LDA     T3
            MOV     B, A
            LDA     M3
            ADC     B
            STA     T3
            RET

; Compare T (u32) with N (u32): CY=1 if T<N, Z=1 if equal, else T>N
T_CMP_N:
            LDA     N3
            MOV     B, A
            LDA     T3
            CMP     B
            JC      TLT
            JZ      TEQ3
            JMP     TGT
TEQ3:       LDA     N2
            MOV     B, A
            LDA     T2
            CMP     B
            JC      TLT
            JZ      TEQ2
            JMP     TGT
TEQ2:       LDA     N1
            MOV     B, A
            LDA     T1
            CMP     B
            JC      TLT
            JZ      TEQ1
            JMP     TGT
TEQ1:       LDA     N0
            MOV     B, A
            LDA     T0
            CMP     B
            JC      TLT
            JZ      TEQ0
            JMP     TGT
TEQ0:       XRA     A
            RET
TLT:        STC
            RET
TGT:        STC
            CMC
            ORA     A
            RET

; ============================================================================
; DATA
; ============================================================================
TITLE:      DB 'Prime generator. (^C to stop)',13,10,'$'
NL:         DB 13,10,'$'

; Current candidate N (u32, little endian)
N0:         DB 0
N1:         DB 0
N2:         DB 0
N3:         DB 0

; Decimal conversion workspace
NW0:        DB 0
NW1:        DB 0
NW2:        DB 0
NW3:        DB 0
OUTBUF:     DS 16                 ; up to 13 chars incl commas + '$'
OUTEND      EQU OUTBUF+15         ; '$' lives here
OUTPTR:     DW 0
PTR32:      DW 0
DIGCNT:     DB 0                  ; NEW: digits since last comma (0..3)

; Square / compare workspace
T0:         DB 0
T1:         DB 0
T2:         DB 0
T3:         DB 0
M0:         DB 0
M1:         DB 0
M2:         DB 0
M3:         DB 0
