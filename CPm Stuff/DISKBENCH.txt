; =====================================================================
; DSKB256.ASM — Disk Speed Benchmark, fixed 256 KB, RTC timed [8080]
; ---------------------------------------------------------------------
; - Requires RomWBW HBIOS + SC727 RTC (DS1302)
; - Writes exactly 256 KB (64 x 4 KB), then reads same amount
; - Uses HBIOS RTCGETTIM (B=20h, RST 1) and HH:MM:SS delta math
; - Progress dot per 4 KB; clear, normal text output
; =====================================================================

BDOS        EQU 0005H
CONOUT      EQU 2
RSTHB       EQU 0008H          ; HBIOS entry (RST 1 vector)

; HBIOS RTCGETTIM:
;   IN : B=20h, HL=TIMBUF (6 bytes)
;   OUT: TIMBUF = [YY,MM,DD,HH,MM,SS] (all BCD), A=status (00=OK)

; ---- Test sizing (constants) ----
CHUNK_RECS   EQU 32            ; 32 * 128 B = 4096 B per dot
TOTAL_CHUNKS EQU 64            ; 64 * 4 KB = 256 KB total

            ORG 0100H

START:
            ; Clear + header
            LXI  D,MSG_HDR
            MVI  C,9
            CALL BDOS

            ; RTC present?
            CALL RTC_READ
            ORA  A
            JZ   RTC_OK
            LXI  D,MSG_RTCERR
            MVI  C,9
            CALL BDOS
            RET
RTC_OK:

            ; Remove any old file
            LXI  D,FCB
            MVI  C,19          ; Delete
            CALL BDOS

; -------------------------
; WRITE 256 KB
; -------------------------
            LXI  D,MSG_MAKE
            MVI  C,9
            CALL BDOS
            LXI  D,FCB
            MVI  C,22          ; Make file
            CALL BDOS

            ; Fill 4 KB buffer with A5h
            MVI  A,0A5H
            LXI  H,BUF
            LXI  B,1000H       ; 4096 bytes
FILLW:      MOV  M,A
            INX  H
            DCX  B
            MOV  A,B
            ORA  C
            JNZ  FILLW

            LXI  D,MSG_WR
            MVI  C,9
            CALL BDOS

            CALL WAIT_NEXT_SEC
            CALL RTC_SNAP_START

            MVI  A,TOTAL_CHUNKS
            STA  OUTER_CT

W_OUTER:
            ; write one 4 KB (32 records)
            LXI  H,BUF
            MVI  A,CHUNK_RECS
            STA  REC_CT
W_INNER:
            ; DMA = HL
            MOV  D,H
            MOV  E,L
            MVI  C,26          ; Set DMA
            CALL BDOS
            ; Write sequential
            LXI  D,FCB
            MVI  C,21          ; Write Seq
            CALL BDOS
            ; HL += 128
            MOV  A,L
            ADI 080H
            MOV  L,A
            JNC  W_NOCRY
            INR  H
W_NOCRY:
            LDA  REC_CT
            DCR  A
            STA  REC_CT
            JNZ  W_INNER

            ; progress dot
            MVI  E,'.'
            CALL PCHAR

            ; next chunk
            LDA  OUTER_CT
            DCR  A
            STA  OUTER_CT
            JNZ  W_OUTER

            ; close file
            LXI  D,FCB
            MVI  C,16          ; Close
            CALL BDOS

            ; time & report
            CALL CRLF
            LXI  D,MSG_WT
            MVI  C,9
            CALL BDOS

            CALL RTC_DELTA_SEC_HL
            SHLD SEC_WR

            ; print seconds (and "(<1s)" if zero)
            CALL PRINT_U16_HL
            LHLD SEC_WR
            MOV  A,H
            ORA  L
            JNZ  WT_SECS_OK
            LXI  D,MSG_LT1
            MVI  C,9
            CALL BDOS
WT_SECS_OK:
            CALL SP_PAD
            LXI  D,MSG_S
            MVI  C,9
            CALL BDOS

            ; KB/s = 256 / seconds  (use 1 if 0 to avoid div/0)
            LHLD SEC_WR
            MOV  A,H
            ORA  L
            JNZ  WT_HASSEC
            INX  H                  ; treat as 1s if 0
WT_HASSEC:  MOV  D,H
            MOV  E,L                ; DE = seconds
            LXI  H,0100H            ; HL = 256 KB
            CALL U16_DIV            ; HL = KB/s
            CALL SP_PAD
            LXI  D,MSG_KBPS
            MVI  C,9
            CALL BDOS
            CALL CRLF

; -------------------------
; READ 256 KB
; -------------------------
            LXI  D,MSG_RD
            MVI  C,9
            CALL BDOS

            LXI  D,FCB
            MVI  C,15          ; Open
            CALL BDOS

            CALL WAIT_NEXT_SEC
            CALL RTC_SNAP_START

            MVI  A,TOTAL_CHUNKS
            STA  OUTER_CT

R_OUTER:
            LXI  H,BUF
            MVI  A,CHUNK_RECS
            STA  REC_CT
R_INNER:
            MOV  D,H
            MOV  E,L
            MVI  C,26          ; Set DMA
            CALL BDOS
            ; Read sequential
            LXI  D,FCB
            MVI  C,20          ; Read Seq
            CALL BDOS
            ; HL += 128
            MOV  A,L
            ADI 080H
            MOV  L,A
            JNC  R_NOCRY
            INR  H
R_NOCRY:
            LDA  REC_CT
            DCR  A
            STA  REC_CT
            JNZ  R_INNER

            ; dot and next chunk
            MVI  E,'.'
            CALL PCHAR
            LDA  OUTER_CT
            DCR  A
            STA  OUTER_CT
            JNZ  R_OUTER

            ; close
            LXI  D,FCB
            MVI  C,16          ; Close
            CALL BDOS

            ; time & report
            CALL CRLF
            LXI  D,MSG_RT
            MVI  C,9
            CALL BDOS

            CALL RTC_DELTA_SEC_HL
            SHLD SEC_RD

            CALL PRINT_U16_HL
            LHLD SEC_RD
            MOV  A,H
            ORA  L
            JNZ  RT_SECS_OK
            LXI  D,MSG_LT1
            MVI  C,9
            CALL BDOS
RT_SECS_OK:
            CALL SP_PAD
            LXI  D,MSG_S
            MVI  C,9
            CALL BDOS

            LHLD SEC_RD
            MOV  A,H
            ORA  L
            JNZ  RT_HASSEC
            INX  H
RT_HASSEC:  MOV  D,H
            MOV  E,L                ; DE = seconds
            LXI  H,0100H            ; HL = 256
            CALL U16_DIV
            CALL SP_PAD
            LXI  D,MSG_KBPS
            MVI  C,9
            CALL BDOS
            CALL CRLF

            ; cleanup
            LXI  D,MSG_CLEAN
            MVI  C,9
            CALL BDOS
            LXI  D,FCB
            MVI  C,19          ; Delete
            CALL BDOS
            LXI  D,MSG_DONE
            MVI  C,9
            CALL BDOS
            RET

; =====================================================================
; RTC helpers
; =====================================================================

RTC_READ:                           ; A=0 on success
            LXI  H,TIMBUF
            MVI  B,020H
            RST  1
            RET

RTC_SNAP_START:                     ; save start HH:MM:SS (binary)
            CALL RTC_READ
            ORA  A
            RNZ
            LDA  TIMBUF+3
            CALL BCD2BIN
            STA  S_HH
            LDA  TIMBUF+4
            CALL BCD2BIN
            STA  S_MM
            LDA  TIMBUF+5
            CALL BCD2BIN
            STA  S_SS
            RET

WAIT_NEXT_SEC:
            CALL RTC_READ
            ORA  A
            RNZ
            LDA  TIMBUF+5
            CALL BCD2BIN
            STA  TMP0
WN_LOOP:    CALL RTC_READ
            ORA  A
            RNZ
            LDA  TIMBUF+5
            CALL BCD2BIN
            MOV  B,A
            LDA  TMP0
            CMP  B
            JZ   WN_LOOP
            RET

; HL = elapsed seconds since RTC_SNAP_START (borrow-aware)
RTC_DELTA_SEC_HL:
            CALL RTC_READ
            ORA  A
            RNZ
            ; current HH/MM/SS -> C_*
            LDA  TIMBUF+3
            CALL BCD2BIN
            STA  C_HH
            LDA  TIMBUF+4
            CALL BCD2BIN
            STA  C_MM
            LDA  TIMBUF+5
            CALL BCD2BIN
            STA  C_SS

            ; seconds
            LDA  C_SS
            MOV  B,A
            LDA  S_SS
            MOV  C,A
            MOV  A,B
            CMP  C
            JNC  DSEC_NO_B
            ADI  60
            SUB  C
            STA  D_SEC
            MVI  A,1
            STA  BORR1
            JMP  DSEC_OK
DSEC_NO_B:  SUB  C
            STA  D_SEC
            XRA  A
            STA  BORR1
DSEC_OK:
            ; minutes (apply seconds borrow)
            LDA  C_MM
            MOV  B,A
            LDA  BORR1
            ORA  A
            JZ   MIN_SKIP_DEC
            MOV  A,B
            ORA  A
            JNZ  MIN_DEC1
            MVI  B,59
            MVI  A,1
            STA  BORR2
            JMP  MIN_ADJ_OK
MIN_DEC1:   DCR  B
            XRA  A
            STA  BORR2
MIN_ADJ_OK:
MIN_SKIP_DEC:
            LDA  S_MM
            MOV  C,A
            MOV  A,B
            CMP  C
            JNC  DMIN_NO_B
            ADI  60
            SUB  C
            STA  D_MIN
            MVI  A,1
            STA  BORR2
            JMP  DMIN_OK
DMIN_NO_B:  SUB  C
            STA  D_MIN
DMIN_OK:
            ; hours (apply minute borrow)
            LDA  C_HH
            MOV  B,A
            LDA  BORR2
            ORA  A
            JZ   H_SKIP_DEC
            MOV  A,B
            ORA  A
            JNZ  H_DEC1
            MVI  B,23
            JMP  H_ADJ_OK
H_DEC1:     DCR  B
H_ADJ_OK:
H_SKIP_DEC:
            LDA  S_HH
            MOV  C,A
            MOV  A,B
            CMP  C
            JNC  DH_NO_B
            ADI  24
            SUB  C
            STA  D_HR
            JMP  DH_OK
DH_NO_B:    SUB  C
            STA  D_HR
DH_OK:
            ; total seconds = H*3600 + M*60 + S
            XRA  A
            MOV  H,A
            MOV  L,A
            ; hours * 3600 (0x0E10)
            LDA  D_HR
            MOV  B,A
            MVI  D,0EH
            MVI  E,10H
ADD_H3600:  MOV  A,B
            ORA  A
            JZ   AH_DONE
            DAD  D
            DCR  B
            JMP  ADD_H3600
AH_DONE:
            ; minutes * 60
            LDA  D_MIN
            MOV  B,A
            MVI  D,00H
            MVI  E,3CH
ADD_M60:    MOV  A,B
            ORA  A
            JZ   AM_DONE
            DAD  D
            DCR  B
            JMP  ADD_M60
AM_DONE:
            ; + seconds
            XRA  A
            MOV  D,A
            LDA  D_SEC
            MOV  E,A
            DAD  D
            RET

; =====================================================================
; Math / utilities
; =====================================================================

; A(BCD) -> A(binary)
BCD2BIN:
            MOV  D,A
            ANI  0FH
            MOV  E,A
            MOV  A,D
            ANI  0F0H
            RRC
            RRC
            RRC
            RRC
            MOV  B,A
            MOV  A,B
            ADD  A              ; 2t
            MOV  C,A
            MOV  A,B
            ADD  A              ; 2t
            ADD  A              ; 4t
            ADD  A              ; 8t
            ADD  C              ; 10t
            ADD  E
            RET

; HL / DE → HL (quotient), unsigned (slow but fine)
U16_DIV:
            XRA  A
            MOV  B,A
            MOV  C,A
UD_LP:      MOV  A,H
            CMP  D
            JC   UD_DONE
            JZ   UD_EQ
            JMP  UD_SUB
UD_EQ:      MOV  A,L
            CMP  E
            JC   UD_DONE
UD_SUB:     MOV  A,L
            SUB  E
            MOV  L,A
            MOV  A,H
            SBB  D
            MOV  H,A
            INR  C
            JNZ  UD_LP
            INR  B
            JMP  UD_LP
UD_DONE:    MOV  H,B
            MOV  L,C
            RET

; print unsigned HL in decimal
PRINT_U16_HL:
            MVI  A,0
            STA  LZFLAG
            LXI  D,2710H        ; 10000
            CALL DIGIT_OUT
            LXI  D,03E8H        ; 1000
            CALL DIGIT_OUT
            LXI  D,0064H        ; 100
            CALL DIGIT_OUT
            LXI  D,000AH        ; 10
            CALL DIGIT_OUT
            LXI  D,0001H        ; 1
            CALL DIGIT_OUT
            RET

DIGIT_OUT:
            PUSH B
            PUSH D
            XRA  A
            MOV  C,A
DO_LOOP:
            MOV  A,H
            CMP  D
            JC   DO_END
            JZ   DO_EQ
            JMP  DO_SUB
DO_EQ:      MOV  A,L
            CMP  E
            JC   DO_END
DO_SUB:     MOV  A,L
            SUB  E
            MOV  L,A
            MOV  A,H
            SBB  D
            MOV  H,A
            INR  C
            JMP  DO_LOOP
DO_END:
            LDA  LZFLAG
            ORA  C
            JNZ  DO_PRINT
            MOV  A,D
            ORA  E
            CPI  01H
            JNZ  DO_SKIP
DO_PRINT:   MVI  A,1
            STA  LZFLAG
            MOV  A,C
            ADI  '0'
            MOV  E,A
            CALL PCHAR
DO_SKIP:
            POP  D
            POP  B
            RET

; BDOS helpers
PCHAR:      MVI  C,CONOUT
            CALL BDOS
            RET

CRLF:       MVI  E,13
            CALL PCHAR
            MVI  E,10
            CALL PCHAR
            RET

SP_PAD:     MVI  E,' '
            CALL PCHAR
            RET

; =====================================================================
; Data
; =====================================================================

OUTER_CT:   DB   00H
REC_CT:     DB   00H
LZFLAG:     DB   00H

SEC_WR:     DW   0000H
SEC_RD:     DW   0000H

; RTC scratch
TIMBUF:     DS   6              ; YY MM DD HH MM SS
S_HH:       DB   00H
S_MM:       DB   00H
S_SS:       DB   00H
C_HH:       DB   00H
C_MM:       DB   00H
C_SS:       DB   00H
D_HR:       DB   00H
D_MIN:      DB   00H
D_SEC:      DB   00H
BORR1:      DB   00H
BORR2:      DB   00H
TMP0:       DB   00H

; FCB for DSKBENCH.DAT
FCB:        DB   0
            DB   'D','S','K','B','E','N','C','H'
            DB   'D','A','T'
            DS   20

; messages
MSG_HDR:    DB  1BH,'[','2','J',1BH,'[','H','$'
            DB  'Disk Speed Benchmark — 256 KB (RTC)',13,10,'$'
MSG_RTCERR: DB  'RTC not available (HBIOS RTCGETTIM failed).',13,10
            DB  'SC727 RTC is required.',13,10,'$'

MSG_MAKE:   DB  'Creating DSKBENCH.DAT ...',13,10,'$'
MSG_WR:     DB  'Write test: 256 KB',13,10,'$'
MSG_WT:     DB  13,10,'Write time: ','$'
MSG_RD:     DB  'Read test : 256 KB',13,10,'$'
MSG_RT:     DB  13,10,'Read time : ','$'
MSG_LT1:    DB  ' (<1s)','$'
MSG_S:      DB  ' s  ','$'
MSG_KBPS:   DB  'KB/s',13,10,'$'
MSG_CLEAN:  DB  'Cleaning up ...',13,10,'$'
MSG_DONE:   DB  'Done.',13,10,'$'

; 4 KB buffer
BUF:        DS   4096

            END  START
