; =====================================================================
; DSKBRTC.ASM — Disk Speed Benchmark using RTC (SC727)  [8080-only]
; ---------------------------------------------------------------------
; - Requires RomWBW HBIOS + SC727 (DS1302) RTC
; - Uses HBIOS RTCGETTIM (B=20h, RST 1) to time >= 1s per phase
; - Progress: one dot per 4 KB written/read
; - Reports seconds and KB/s, then deletes the test file
; - Strict Intel 8080 mnemonics; ASM.COM compatible
; =====================================================================

BDOS        EQU 0005H
CONOUT      EQU 2
RSTHB       EQU 0008H          ; HBIOS entry (RST 1 vector)

; HBIOS RTCGETTIM:
;   IN : B=20h, HL=TIMBUF (6 bytes)
;   OUT: TIMBUF = [YY,MM,DD,HH,MM,SS] (BCD), A=status (00=OK)

; ---- Tunables ----
CHUNK_RECS   EQU 32            ; 32 * 128B = 4096 bytes per dot
MAX_CHUNKS   EQU 2048          ; safety cap (~8 MB total if hit)

            ORG 0100H

START:
            ; Clear screen + header
            LXI  D,MSG_HDR
            MVI  C,9
            CALL BDOS

            ; Check RTC
            CALL RTC_READ
            ORA  A
            JZ   RTC_OK
            LXI  D,MSG_RTCERR
            MVI  C,9
            CALL BDOS
            RET
RTC_OK:

            ; Remove any old file
            LXI  D,FCB
            MVI  C,19          ; Delete
            CALL BDOS

; -------------------------
; WRITE PHASE
; -------------------------
            LXI  D,MSG_MAKE
            MVI  C,9
            CALL BDOS
            LXI  D,FCB
            MVI  C,22          ; Make file
            CALL BDOS

            ; Fill 4 KB buffer with A5h
            MVI  A,0A5H
            LXI  H,BUF
            LXI  B,1000H       ; 4096 bytes
FILLW:      MOV  M,A
            INX  H
            DCX  B
            MOV  A,B
            ORA  C
            JNZ  FILLW

            LXI  D,MSG_WR
            MVI  C,9
            CALL BDOS

            CALL WAIT_NEXT_SEC      ; align to a new second
            CALL RTC_SNAP_START     ; remember start HH:MM:SS
            XRA  A
            STA  WR_LO
            STA  WR_HI              ; chunks written = 0

W_LOOP:
            ; write one 4 KB chunk (32 records)
            LXI  H,BUF
            MVI  A,CHUNK_RECS
            STA  REC_CT
W_INNER:    ; DMA = HL
            MOV  D,H
            MOV  E,L
            MVI  C,26              ; Set DMA
            CALL BDOS
            ; Write sequential
            LXI  D,FCB
            MVI  C,21              ; Write Seq
            CALL BDOS
            ; HL += 128
            MOV  A,L
            ADI 080H
            MOV  L,A
            JNC  W_NOCR
            INR  H
W_NOCR:
            LDA  REC_CT
            DCR  A
            STA  REC_CT
            JNZ  W_INNER

            ; dot + chunk++
            MVI  E,'.'
            CALL PCHAR
            CALL INC_WR_CHK

            ; stop when elapsed >= 1s
            CALL RTC_DELTA_SEC_HL   ; HL = elapsed seconds
            MOV  A,H
            ORA  L
            JNZ  W_DONE

            ; safety cap: if chunks >= MAX_CHUNKS → stop
            LHLD WR_LO             ; HL = chunks
            LXI  D,MAX_CHUNKS
            MOV  A,H
            CMP  D
            JC   W_CONT
            JZ   W_EQH
            JMP  W_DONE
W_EQH:      MOV  A,L
            CMP  E
            JC   W_CONT
            JMP  W_DONE

W_CONT:     JMP  W_LOOP

W_DONE:
            ; Close file after writing
            LXI  D,FCB
            MVI  C,16              ; Close
            CALL BDOS

            ; Report write results
            CALL CRLF
            LXI  D,MSG_WT
            MVI  C,9
            CALL BDOS
            CALL RTC_DELTA_SEC_HL
            SHLD SEC_WR
            CALL PRINT_U16_HL
            CALL SP_PAD
            LXI  D,MSG_S
            MVI  C,9
            CALL BDOS

            ; KB/s = (chunks * 4) / seconds
            LHLD SEC_WR
            MOV  A,H
            ORA  L
            JNZ  W_HASSEC
            INX  H                      ; avoid div/0
W_HASSEC:   MOV  D,H
            MOV  E,L                    ; DE = seconds
            LHLD WR_LO                  ; HL = chunks
            DAD  H                      ; HL *= 2
            DAD  H                      ; HL *= 4  (KB processed)
            CALL U16_DIV                ; HL = KB/s
            CALL SP_PAD
            LXI  D,MSG_KBPS
            MVI  C,9
            CALL BDOS
            CALL CRLF

; -------------------------
; READ PHASE
; -------------------------
            LXI  D,MSG_RD
            MVI  C,9
            CALL BDOS

            ; Re-open file for reading
            LXI  D,FCB
            MVI  C,15                  ; Open
            CALL BDOS

            XRA  A
            STA  RD_LO
            STA  RD_HI

            CALL WAIT_NEXT_SEC
            CALL RTC_SNAP_START

R_LOOP:
            LXI  H,BUF
            MVI  A,CHUNK_RECS
            STA  REC_CT

R_INNER:
            MOV  D,H
            MOV  E,L
            MVI  C,26
            CALL BDOS
            ; Read sequential
            LXI  D,FCB
            MVI  C,20                  ; Read Seq
            CALL BDOS
            CPI  01H
            JNZ  R_OK
            ; EOF: close & reopen, continue
            LXI  D,FCB
            MVI  C,16                  ; Close
            CALL BDOS
            LXI  D,FCB
            MVI  C,15                  ; Open
            CALL BDOS
            JMP  R_INNER_CONT
R_OK:
            ; advance DMA
            MOV  A,L
            ADI 080H
            MOV  L,A
            JNC  R_NOCR
            INR  H
R_NOCR:
R_INNER_CONT:
            LDA  REC_CT
            DCR  A
            STA  REC_CT
            JNZ  R_INNER

            ; dot + chunk++
            MVI  E,'.'
            CALL PCHAR
            CALL INC_RD_CHK

            ; stop when elapsed >= 1s
            CALL RTC_DELTA_SEC_HL
            MOV  A,H
            ORA  L
            JNZ  R_DONE

            ; optional cap like write
            LHLD RD_LO
            LXI  D,MAX_CHUNKS
            MOV  A,H
            CMP  D
            JC   R_LOOP
            JZ   R_EQH
            JMP  R_DONE
R_EQH:      MOV  A,L
            CMP  E
            JC   R_LOOP
            JMP  R_DONE

R_DONE:
            ; close file after reading
            LXI  D,FCB
            MVI  C,16
            CALL BDOS

            ; Report read results
            CALL CRLF
            LXI  D,MSG_RT
            MVI  C,9
            CALL BDOS
            CALL RTC_DELTA_SEC_HL
            SHLD SEC_RD
            CALL PRINT_U16_HL
            CALL SP_PAD
            LXI  D,MSG_S
            MVI  C,9
            CALL BDOS

            LHLD SEC_RD
            MOV  A,H
            ORA  L
            JNZ  R_HASSEC
            INX  H
R_HASSEC:   MOV  D,H
            MOV  E,L                    ; DE = seconds
            LHLD RD_LO                  ; HL = chunks
            DAD  H                      ; *2
            DAD  H                      ; *4
            CALL U16_DIV
            CALL SP_PAD
            LXI  D,MSG_KBPS
            MVI  C,9
            CALL BDOS
            CALL CRLF

            ; ----- CLEANUP -----
            LXI  D,MSG_CLEAN
            MVI  C,9
            CALL BDOS
            LXI  D,FCB
            MVI  C,19                  ; Delete file
            CALL BDOS
            LXI  D,MSG_DONE
            MVI  C,9
            CALL BDOS
            RET

; =====================================================================
; RTC helpers
; =====================================================================

; RTC_READ: Fill TIMBUF via HBIOS (A=0 on success)
RTC_READ:
            LXI  H,TIMBUF
            MVI  B,020H
            RST  1
            RET

; RTC_SNAP_START: store start HH/MM/SS (binary) as S_HH,S_MM,S_SS
RTC_SNAP_START:
            CALL RTC_READ
            ORA  A
            RNZ
            LDA  TIMBUF+3
            CALL BCD2BIN
            STA  S_HH
            LDA  TIMBUF+4
            CALL BCD2BIN
            STA  S_MM
            LDA  TIMBUF+5
            CALL BCD2BIN
            STA  S_SS
            RET

; WAIT_NEXT_SEC: block until RTC seconds changes
WAIT_NEXT_SEC:
            CALL RTC_READ
            ORA  A
            RNZ
            LDA  TIMBUF+5
            CALL BCD2BIN
            STA  TMP0
WN_LOOP:    CALL RTC_READ
            ORA  A
            RNZ
            LDA  TIMBUF+5
            CALL BCD2BIN
            MOV  B,A
            LDA  TMP0
            CMP  B
            JZ   WN_LOOP
            RET

; RTC_DELTA_SEC_HL
; HL = elapsed seconds since RTC_SNAP_START (0..65535),
; handles wrap across minute/hour (and midnight).
RTC_DELTA_SEC_HL:
            CALL RTC_READ
            ORA  A
            RNZ
            ; current HH/MM/SS -> C_HH,C_MM,C_SS (binary)
            LDA  TIMBUF+3
            CALL BCD2BIN
            STA  C_HH
            LDA  TIMBUF+4
            CALL BCD2BIN
            STA  C_MM
            LDA  TIMBUF+5
            CALL BCD2BIN
            STA  C_SS

            ; ---- seconds with borrow into minutes ----
            LDA  C_SS           ; curS
            MOV  B,A
            LDA  S_SS           ; startS
            MOV  C,A
            MOV  A,B
            CMP  C
            JNC  DSEC_NO_BORR
            ADI  60
            SUB  C
            STA  D_SEC
            MVI  A,1
            STA  BORR1
            JMP  DSEC_DONE
DSEC_NO_BORR:
            SUB  C
            STA  D_SEC
            XRA  A
            STA  BORR1
DSEC_DONE:

            ; ---- minutes (apply seconds-borrow first) ----
            LDA  C_MM           ; curM
            MOV  B,A
            LDA  BORR1
            ORA  A
            JZ   MIN_NO_ADJ
            MOV  A,B
            ORA  A
            JNZ  MIN_DEC1
            MVI  B,59
            MVI  A,1
            STA  BORR2
            JMP  MIN_ADJ_OK
MIN_DEC1:   DCR  B
            XRA  A
            STA  BORR2
MIN_ADJ_OK:
MIN_NO_ADJ:
            LDA  S_MM
            MOV  C,A            ; startM
            MOV  A,B            ; curM adjusted
            CMP  C
            JNC  DMIN_NO_BORR
            ADI  60
            SUB  C
            STA  D_MIN
            MVI  A,1
            STA  BORR2
            JMP  DMIN_DONE
DMIN_NO_BORR:
            SUB  C
            STA  D_MIN
DMIN_DONE:

            ; ---- hours (apply minute-borrow) ----
            LDA  C_HH           ; curH
            MOV  B,A
            LDA  BORR2
            ORA  A
            JZ   HOUR_NO_ADJ
            MOV  A,B
            ORA  A
            JNZ  HOUR_DEC1
            MVI  B,23
            JMP  HOUR_ADJ_OK
HOUR_DEC1:  DCR  B
HOUR_ADJ_OK:
HOUR_NO_ADJ:
            LDA  S_HH
            MOV  C,A            ; startH
            MOV  A,B            ; curH adjusted
            CMP  C
            JNC  DHOUR_NO_BORR
            ADI  24
            SUB  C
            STA  D_HR
            JMP  DHOUR_DONE
DHOUR_NO_BORR:
            SUB  C
            STA  D_HR
DHOUR_DONE:

            ; ---- total seconds = H*3600 + M*60 + S ----
            XRA  A
            MOV  H,A
            MOV  L,A

            ; add hours * 3600 (0x0E10)
            LDA  D_HR
            MOV  B,A
            MVI  D,0EH
            MVI  E,10H
ADD_H3600:  MOV  A,B
            ORA  A
            JZ   AH_DONE
            DAD  D
            DCR  B
            JMP  ADD_H3600
AH_DONE:
            ; add minutes * 60 (0x003C)
            LDA  D_MIN
            MOV  B,A
            MVI  D,00H
            MVI  E,3CH
ADD_M60:    MOV  A,B
            ORA  A
            JZ   AM_DONE
            DAD  D
            DCR  B
            JMP  ADD_M60
AM_DONE:
            ; add seconds (0..59)
            XRA  A
            MOV  D,A
            LDA  D_SEC
            MOV  E,A
            DAD  D
            RET

; =====================================================================
; Math/utility helpers
; =====================================================================

; BCD2BIN: A (BCD 00..99) -> A (binary)
BCD2BIN:
            MOV  D,A
            ANI  0FH
            MOV  E,A            ; ones
            MOV  A,D
            ANI  0F0H
            RRC
            RRC
            RRC
            RRC                 ; tens
            MOV  B,A            ; B=tens
            ; 10*tens = (t<<1) + (t<<3)
            MOV  A,B
            ADD  A              ; 2t
            MOV  C,A
            MOV  A,B
            ADD  A              ; 2t
            ADD  A              ; 4t
            ADD  A              ; 8t
            ADD  C              ; 10t
            ADD  E              ; + ones
            RET

; U16_DIV — HL / DE (unsigned), quotient -> HL
U16_DIV:
            XRA  A
            MOV  B,A
            MOV  C,A            ; BC=0
UD_LP:      MOV  A,H
            CMP  D
            JC   UD_DONE
            JZ   UD_EQ
            JMP  UD_SUB
UD_EQ:      MOV  A,L
            CMP  E
            JC   UD_DONE
UD_SUB:     MOV  A,L
            SUB  E
            MOV  L,A
            MOV  A,H
            SBB  D
            MOV  H,A
            INR  C
            JNZ  UD_LP
            INR  B
            JMP  UD_LP
UD_DONE:    MOV  H,B
            MOV  L,C
            RET

; PRINT_U16_HL — print unsigned HL in decimal (no leading zeros)
PRINT_U16_HL:
            MVI  A,0
            STA  LZFLAG

            LXI  D,2710H        ; 10000
            CALL DIGIT_OUT
            LXI  D,03E8H        ; 1000
            CALL DIGIT_OUT
            LXI  D,0064H        ; 100
            CALL DIGIT_OUT
            LXI  D,000AH        ; 10
            CALL DIGIT_OUT
            LXI  D,0001H        ; 1
            CALL DIGIT_OUT
            RET

; DIGIT_OUT: DE=place, HL=value → prints one digit, leaves HL=remainder
DIGIT_OUT:
            PUSH B
            PUSH D
            XRA  A
            MOV  C,A            ; C = digit
DO_LOOP:
            ; if HL < DE then end
            MOV  A,H
            CMP  D
            JC   DO_END
            JZ   DO_EQ
            JMP  DO_SUB
DO_EQ:      MOV  A,L
            CMP  E
            JC   DO_END
DO_SUB:     ; HL -= DE
            MOV  A,L
            SUB  E
            MOV  L,A
            MOV  A,H
            SBB  D
            MOV  H,A
            INR  C
            JMP  DO_LOOP
DO_END:
            ; print digit if not leading, else skip unless last place
            LDA  LZFLAG
            ORA  C
            JNZ  DO_PRINT
            ; if DE==0001 (last place), must print '0'
            MOV  A,D
            ORA  E
            CPI  01H
            JNZ  DO_SKIP
DO_PRINT:   MVI  A,1
            STA  LZFLAG
            MOV  A,C
            ADI  '0'
            MOV  E,A
            CALL PCHAR
DO_SKIP:
            POP  D
            POP  B
            RET

; BDOS helpers
PCHAR:      MVI  C,CONOUT
            CALL BDOS
            RET

CRLF:       MVI  E,13
            CALL PCHAR
            MVI  E,10
            CALL PCHAR
            RET

SP_PAD:     MVI  E,' '
            CALL PCHAR
            RET

; counters
INC_WR_CHK:
            LHLD WR_LO
            INX  H
            SHLD WR_LO
            RET

INC_RD_CHK:
            LHLD RD_LO
            INX  H
            SHLD RD_LO
            RET

; =====================================================================
; Data
; =====================================================================

REC_CT:     DB   00H
WR_LO:      DW   0000H
WR_HI:      DW   0000H
RD_LO:      DW   0000H
RD_HI:      DW   0000H
LZFLAG:     DB   00H

; RTC scratch
TIMBUF:     DS   6              ; YY MM DD HH MM SS (BCD)
S_HH:       DB   00H
S_MM:       DB   00H
S_SS:       DB   00H
C_HH:       DB   00H
C_MM:       DB   00H
C_SS:       DB   00H
D_HR:       DB   00H
D_MIN:      DB   00H
D_SEC:      DB   00H
BORR1:      DB   00H
BORR2:      DB   00H
TMP0:       DB   00H

SEC_WR:     DW   0000H
SEC_RD:     DW   0000H

; FCB for DSKBENCH.DAT
FCB:        DB   0              ; current drive
            DB   'D','S','K','B','E','N','C','H'
            DB   'D','A','T'
            DS   20

; messages
MSG_HDR:    DB  1BH,'[','2','J',1BH,'[','H','$'
            DB  'Disk Speed Benchmark (RTC)',13,10,'$'
MSG_RTCERR: DB  'RTC not available (HBIOS RTCGETTIM failed).',13,10
            DB  'This test requires the SC727 RTC.',13,10,'$'

MSG_MAKE:   DB  'Creating DSKBENCH.DAT ...',13,10,'$'
MSG_WR:     DB  'Write test (auto >= 1s):',13,10,'$'
MSG_WT:     DB  13,10,'Write time: ','$'
MSG_RD:     DB  'Read test  (auto >= 1s):',13,10,'$'
MSG_RT:     DB  13,10,'Read time:  ','$'
MSG_S:      DB  ' s  ','$'
MSG_KBPS:   DB  'KB/s',13,10,'$'
MSG_CLEAN:  DB  'Cleaning up ...',13,10,'$'
MSG_DONE:   DB  'Done.',13,10,'$'

; 4 KB buffer
BUF:        DS   4096

            END  START
