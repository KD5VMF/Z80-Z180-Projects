; TIME.ASM — 8080 mnemonics, CP/M + RomWBW HBIOS
; Prints YYYY-MM-DD HH:MM:SS once per second

BDOS    EQU 0005H
CONOUT  EQU 2

            ORG 0100H

START:      CALL CLS
MAIN:       CALL READRTC              ; fills TIMBUF (BCD: YY MM DD HH MM SS)
            CALL PRINTDT
            CALL DELAY1S
            JMP  MAIN

; --- HBIOS: RTCGETTIM (B=20h, HL=buffer), entry at 0008h ---
READRTC:    LXI  H,TIMBUF
            MVI  B,020H
            RST  1
            RET

; --- Print YYYY-MM-DD HH:MM:SS ---
PRINTDT:    MVI  E,'2'                ; "20"
            CALL PCHAR
            MVI  E,'0'
            CALL PCHAR

            LDA  TIMBUF+0             ; YY
            CALL P2DIG
            MVI  E,'-'
            CALL PCHAR

            LDA  TIMBUF+1             ; MM
            CALL P2DIG
            MVI  E,'-'
            CALL PCHAR

            LDA  TIMBUF+2             ; DD
            CALL P2DIG
            MVI  E,' '
            CALL PCHAR

            LDA  TIMBUF+3             ; HH
            CALL P2DIG
            MVI  E,':'
            CALL PCHAR

            LDA  TIMBUF+4             ; MM
            CALL P2DIG
            MVI  E,':'
            CALL PCHAR

            LDA  TIMBUF+5             ; SS
            CALL P2DIG
            MVI  E,13                 ; CR
            CALL PCHAR
            MVI  E,10                 ; LF
            CALL PCHAR
            RET

; --- A=BCD -> print two ASCII digits (store to mem so BDOS can't clobber) ---
P2DIG:      CALL BCD2ASC              ; H=tens ASCII, L=ones ASCII
            MOV  A,H
            STA  DIGTMP               ; save tens
            MOV  A,L
            STA  DIGTMP+1             ; save ones
            LDA  DIGTMP
            MOV  E,A
            CALL PCHAR
            LDA  DIGTMP+1
            MOV  E,A
            CALL PCHAR
            RET

; --- BCD -> ASCII pair in H,L ---
BCD2ASC:    MOV  D,A
            ANI  00FH                 ; ones
            ADI  '0'
            MOV  L,A
            MOV  A,D
            ANI  0F0H                 ; tens
            RRC
            RRC
            RRC
            RRC
            ADI  '0'
            MOV  H,A
            RET

; --- BDOS char out (E) ---
PCHAR:      MVI  C,CONOUT
            CALL BDOS
            RET

CLS:        MVI  E,26
            CALL PCHAR
            RET

DELAY1S:    LXI  H,0FFFFH             ; crude ~1s at ~7.37 MHz
DLY1:       DCX  H
            MOV  A,H
            ORA  L
            JNZ  DLY1
            RET

TIMBUF:     DS   6                    ; YY MM DD HH MM SS (BCD)
DIGTMP:     DS   2                    ; temp for two ASCII digits

            END  START
