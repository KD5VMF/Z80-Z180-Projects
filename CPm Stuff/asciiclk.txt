; ==============================================================
; BIGCLOCK.ASM — 8080 mnemonics, CP/M + RomWBW HBIOS + SC719
; --------------------------------------------------------------
; Features:
;   * Reads RTC (SC727 module) through RomWBW HBIOS (no direct RTC port I/O)
;   * Renders big "HH:MM:SS" at the top-left using a 5×7 ASCII font
;   * Updates once per second (waits for the second to change)
;   * Mirrors seconds (0..59) to SC719 LED latch at a configurable I/O port
;   * Q or q quits (clears LEDs and restores cursor)
;
; I/O / Hardware overview:
;   - RTC (SC727) actual I/O base: C0h (as shown by HBIOS banner)
;       BUT: user programs should NOT touch C0h directly under RomWBW.
;       Use HBIOS call: RST 1 with B=20h (RTCGETTIM) and HL=buffer.
;   - LEDs (SC719): 8-bit output latch at base port set by JP1 jumpers.
;       Default (no jumpers) is often 00h. We OUT the seconds byte there.
; ==============================================================

BDOS        EQU 0005H          ; CP/M BDOS entry
CONOUT      EQU 2              ; BDOS function 2: console output (char in E)

; ====== CONFIGURE YOUR SC719 I/O BASE ADDRESS & POLARITY ======
DIO_BASE    EQU 00H            ; <-- change to your SC719 JP1 base if not 00h
INVERT_XOR  EQU 00H            ; 00h if LEDs are active-HIGH, FFh if active-LOW

            ORG 0100H          ; CP/M .COM entry point

; --------------------------------------------------------------
; Program entry & initial UI setup
; --------------------------------------------------------------
START:
            ; Clear screen, home cursor, hide cursor (ANSI sequences via BDOS 9)
            LXI  D,ESC2J
            MVI  C,9
            CALL BDOS
            LXI  D,HOME
            MVI  C,9
            CALL BDOS
            LXI  D,HIDECUR
            MVI  C,9
            CALL BDOS

            ; SC719 LED self-test: verify address & polarity
            MVI  A,055H        ; 01010101
            CALL LEDOUT
            CALL SHORTDLY
            MVI  A,0AAH        ; 10101010
            CALL LEDOUT
            CALL SHORTDLY
            MVI  A,0FFH        ; all on
            CALL LEDOUT
            CALL SHORTDLY
            XRA  A             ; all off
            CALL LEDOUT
            CALL SHORTDLY

; --------------------------------------------------------------
; Main loop: draw big time, update LEDs, wait for next tick
; --------------------------------------------------------------
MAIN:
            ; Always start drawing from the top-left
            LXI  D,HOME
            MVI  C,9
            CALL BDOS

            CALL READRTC       ; TIMBUF = YY,MM,DD,HH,MM,SS (all BCD)
            LDA  TIMBUF+5
            STA  SEC_PREV      ; remember current seconds (BCD) for tick wait

            CALL MAKEBIGSTR    ; BIGSTR[0..7] = "HH:MM:SS" (ASCII chars)
            CALL PRINTBIGTIME  ; render large 5×7 digits
            CALL WRITELED      ; mirror seconds to SC719 LEDs at DIO_BASE

; Wait until the second changes, or 'Q' to quit
WAITNEXT:   CALL KEYCHECK
            ORA  A
            JZ   WN1
            ANI  05FH          ; normalize to uppercase
            CPI  'Q'
            JZ   QUIT
WN1:        CALL SHORTDLY
            CALL READRTC
            LDA  TIMBUF+5      ; A = current seconds (BCD)
            MOV  D,A
            LDA  SEC_PREV
            CMP  D
            JZ   WAITNEXT      ; still the same second
            MOV  A,D
            STA  SEC_PREV
            JMP  MAIN

; --------------------------------------------------------------
; Quit: clear LEDs, restore cursor, newline, return to CP/M
; --------------------------------------------------------------
QUIT:       XRA  A
            CALL LEDOUT
            LXI  D,SHOWCUR
            MVI  C,9
            CALL BDOS
            MVI  E,13          ; CR
            CALL PCHAR
            MVI  E,10          ; LF
            CALL PCHAR
            RET

; ==============================================================
; HBIOS RTC access (RomWBW)
;   RTCGETTIM: B=20h, HL=buffer, RST 1 (vector 0008h)
;   Returns YY,MM,DD,HH,MM,SS in BCD to the buffer at HL
; ==============================================================
READRTC:    LXI  H,TIMBUF
            MVI  B,020H
            RST  1
            RET

; ==============================================================
; MAKEBIGSTR — build ASCII "HH:MM:SS" into BIGSTR[0..7]
; ==============================================================
MAKEBIGSTR:
            ; Hours (TIMBUF+3)
            LDA  TIMBUF+3
            CALL BCD2ASC
            MOV  A,H
            STA  BIGSTR+0      ; hour tens ASCII
            MOV  A,L
            STA  BIGSTR+1      ; hour ones ASCII
            MVI  A,':'
            STA  BIGSTR+2

            ; Minutes (TIMBUF+4)
            LDA  TIMBUF+4
            CALL BCD2ASC
            MOV  A,H
            STA  BIGSTR+3
            MOV  A,L
            STA  BIGSTR+4
            MVI  A,':'
            STA  BIGSTR+5

            ; Seconds (TIMBUF+5)
            LDA  TIMBUF+5
            CALL BCD2ASC
            MOV  A,H
            STA  BIGSTR+6
            MOV  A,L
            STA  BIGSTR+7
            RET

; ==============================================================
; PRINTBIGTIME — render BIGSTR[0..7] using 5×7 FONT
;   For each of 7 rows:
;     for each of 8 chars: get its glyph row byte, print 5 pixels
;     then CR/LF
; Pixel rule: bit7..bit3 are 5 pixels (1='#', 0=' ')
; ==============================================================
PRINTBIGTIME:
            MVI  A,00H
            STA  ROWI          ; row index = 0
PR_ROW:
            ; col loop init
            MVI  A,00H
            STA  COLI          ; col index = 0
PR_COL:
            ; Load ASCII char ch = BIGSTR[COLI]
            ; HL = BIGSTR + COLI
            LXI  H,BIGSTR
            LDA  COLI
            MOV  E,A
            MVI  D,00H
            DAD  D             ; HL += (00:COLI)
            MOV  A,M           ; A = ch

            ; Convert ASCII to glyph index B (0..9 digits, 10=':', 11=blank)
            CALL CHARTOGLYPH    ; returns B

            ; Get pattern byte for current row (E=row)
            LDA  ROWI
            MOV  E,A           ; E = row
            CALL GETROWPAT     ; returns A = pattern bits (bits 7..3)
            MOV  D,A           ; D = pattern (we'll shift it left each pixel)

            ; Print 5 pixels: test bit7, then shift left via ADD A
            MVI  A,05H
            STA  COUNT5
PR_PIX:
            LDA  COUNT5
            ORA  A
            JZ   PR_AFTER_PIX

            MOV  A,D
            ANI  080H
            JZ   PR_SPACE
            MVI  E,'#'
            CALL PCHAR
            JMP  PR_SHIFT
PR_SPACE:   MVI  E,' '
            CALL PCHAR

PR_SHIFT:   ; shift pattern left by 1: D <<= 1  (8080: A=D, ADD A, D=A)
            MOV  A,D
            ADD  A
            MOV  D,A

            ; next pixel in this glyph row
            LDA  COUNT5
            DCR  A
            STA  COUNT5
            JMP  PR_PIX

PR_AFTER_PIX:
            ; inter-character space
            MVI  E,' '
            CALL PCHAR

            ; next character (COLI++)
            LDA  COLI
            INR  A
            STA  COLI
            CPI  08H           ; 8 glyphs total (HH:MM:SS)
            JNZ  PR_COL

            ; end of this row → CR/LF
            MVI  E,13
            CALL PCHAR
            MVI  E,10
            CALL PCHAR

            ; next row (ROWI++)
            LDA  ROWI
            INR  A
            STA  ROWI
            CPI  07H           ; 7 rows total (0..6)
            JNZ  PR_ROW
            RET

; ==============================================================
; CHARTOGLYPH — ASCII → font index
;  Input:  A=ASCII ('0'..'9' or ':')
;  Output: B=index (0..9 digits, 10=':', 11=blank)
; ==============================================================
CHARTOGLYPH:
            CPI  ':'           ; ASCII 3Ah?
            JNZ  CG_CHECKDIG
            MVI  B,0AH         ; index 10 = colon
            RET
CG_CHECKDIG:
            CPI  '0'
            JC   CG_BLANK      ; below '0' → blank
            SUI  '0'           ; A = digit index 0..?
            CPI  0AH
            JNC  CG_BLANK      ; >=10 → blank
            MOV  B,A           ; 0..9 digit
            RET
CG_BLANK:   MVI  B,0BH         ; index 11 = blank
            RET

; ==============================================================
; GETROWPAT — return pattern byte for glyph B at row E
;  Input:  B=glyph index (0..11), E=row (0..6)
;  Output: A=pattern byte (bits 7..3 used as pixels)
; Notes:
;   Uses DE as a working register, so we save the row in memory first.
; ==============================================================
ROWTMP:     DB 00H
GETROWPAT:
            MOV  A,E
            STA  ROWTMP        ; save row (E will be clobbered)
            LXI  H,FONT        ; base of font table
            LXI  D,0007H       ; step size = 7 bytes per glyph
            MOV  C,B           ; repeat C times to add 7 each time
GRP_LOOP:
            MOV  A,C
            ORA  A
            JZ   GRP_DONE
            DAD  D             ; HL += 7
            DCR  C
            JMP  GRP_LOOP
GRP_DONE:
            ; add row offset (0..6)
            MVI  D,00H
            LDA  ROWTMP
            MOV  E,A
            DAD  D             ; HL += row
            MOV  A,M           ; A = pattern byte for this glyph row
            RET

; ==============================================================
; WRITELED — BCD SS → binary 0..59, then write to SC719
; ==============================================================
WRITELED:
            LDA  TIMBUF+5      ; BCD seconds 0xSS
            MOV  D,A
            ANI  00FH          ; ones
            MOV  E,A
            MOV  A,D
            ANI  0F0H          ; tens nibble
            RRC
            RRC
            RRC
            RRC                 ; A = tens (0..5)
            MOV  B,A           ; B = tens
            ; A = tens*10 = (t<<3) + (t<<1)
            MOV  A,B           ; A = t
            ADD  A             ; A = 2t
            MOV  C,A           ; C = 2t
            MOV  A,B           ; A = t
            ADD  A             ; 2t
            ADD  A             ; 4t
            ADD  A             ; 8t
            ADD  C             ; 10t
            ADD  E             ; + ones => 0..59
            ; fallthrough to LEDOUT

LEDOUT:     XRI  INVERT_XOR    ; invert if active-low LEDs
            OUT  DIO_BASE      ; write to SC719 latch @ JP1 base
            RET

; ==============================================================
; KEYCHECK — non-blocking key read (BDOS 6 with E=FFh)
;  Returns A=0 if no key; else A=ASCII
; ==============================================================
KEYCHECK:
            PUSH B
            PUSH D
            MVI  C,6
            MVI  E,0FFH
            CALL BDOS
            MOV  L,A
            POP  D
            POP  B
            MOV  A,L
            RET

; ==============================================================
; Helpers: BCD→ASCII (H=tens ASCII, L=ones ASCII), console char,
;          short delay used for self-test & idle during tick wait
; ==============================================================
BCD2ASC:    MOV  D,A
            ANI  00FH          ; ones
            ADI  '0'
            MOV  L,A
            MOV  A,D
            ANI  0F0H          ; tens
            RRC
            RRC
            RRC
            RRC
            ADI  '0'
            MOV  H,A
            RET

PCHAR:      MVI  C,CONOUT
            CALL BDOS
            RET

SHORTDLY:   LXI  H,2000H
SDLY1:      DCX  H
            MOV  A,H
            ORA  L
            JNZ  SDLY1
            RET

; ==============================================================
; Data: RTC buffer, work bytes, ANSI strings, and FONT table
; ==============================================================
TIMBUF:     DS   6             ; YY,MM,DD,HH,MM,SS (all BCD)
BIGSTR:     DS   8             ; "HH:MM:SS"
ROWI:       DB   00H
COLI:       DB   00H
COUNT5:     DB   00H
SEC_PREV:   DB   00H

; ANSI escape sequences for BDOS 9 (string print, $-terminated)
ESC2J:      DB  1BH,'[','2','J','$'      ; clear screen
HOME:       DB  1BH,'[','H','$'          ; cursor home
HIDECUR:    DB  1BH,'?','2','5','l','$'  ; hide cursor
SHOWCUR:    DB  1BH,'?','2','5','h','$'  ; show cursor

; --------------------------------------------------------------
; 5×7 FONT (12 glyphs × 7 rows)
; Bits used: 7..3 = 5 pixels left→right (1='#', 0=' ')
; Index: 0..9 digits '0'..'9', 10=':', 11=blank
; --------------------------------------------------------------
FONT:
; '0'
            DB 070H,088H,088H,088H,088H,088H,070H
; '1'
            DB 020H,060H,020H,020H,020H,020H,070H
; '2'
            DB 070H,088H,008H,030H,040H,080H,0F8H
; '3'
            DB 070H,088H,008H,030H,008H,088H,070H
; '4'
            DB 010H,030H,050H,090H,0F8H,010H,010H
; '5'
            DB 0F8H,080H,0F0H,008H,008H,088H,070H
; '6'
            DB 070H,080H,0F0H,088H,088H,088H,070H
; '7'
            DB 0F8H,008H,010H,020H,040H,040H,040H
; '8'
            DB 070H,088H,088H,070H,088H,088H,070H
; '9'
            DB 070H,088H,088H,078H,008H,008H,070H
; ':'
            DB 000H,020H,020H,000H,020H,020H,000H
; blank
            DB 000H,000H,000H,000H,000H,000H,000H

            END START
